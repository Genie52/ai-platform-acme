Example policy
Only supply planners can move stock, and only within limits

Applied to every agent behind the planning Gateway, whatever model or framework it uses.

permit(
  principal is AgentCore::OAuthUser,
  action == AgentCore::Action::"SAP__create_stock_transfer",
  resource == AgentCore::Gateway::"<PLANNING_GATEWAY_ARN>"
)
when {
  principal.hasTag("role") &&
  principal.getTag("role") == "supply-planner" &&
  context.input.quantity < 5000
};
